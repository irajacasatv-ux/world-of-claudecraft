import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';

const paths = [
  'docs/screenshots/eastbrook-vale-rebuild/polish/metadata/after-desktop-ultra.json',
  'docs/screenshots/eastbrook-vale-rebuild/polish/metadata/after-mobile-low.json',
  'docs/screenshots/eastbrook-vale-rebuild/polish/performance/after-desktop-ultra-town.json',
  'docs/screenshots/eastbrook-vale-rebuild/polish/performance/after-mobile-low-town.json',
];
function differences(a, b, path = '') {
  if (JSON.stringify(a) === JSON.stringify(b)) return [];
  if (typeof a !== 'object' || a === null || typeof b !== 'object' || b === null) return [path];
  return [...new Set([...Object.keys(a), ...Object.keys(b)])]
    .flatMap((key) => differences(a[key], b[key], `${path}.${key}`));
}
for (const path of paths) {
  const before = JSON.parse(execFileSync('git', ['show', `ff738f61a1:${path}`], { encoding: 'utf8' }));
  const after = JSON.parse(readFileSync(path, 'utf8'));
  const changed = differences(before, after);
  assert.ok(changed.length > 0);
  assert.ok(changed.every((key) => key.endsWith('.polishProvenance.fingerprint')
    || key.endsWith('.polishProvenance.components.runtimeRender.renderer.sha256')));
  console.log(`${path}: ${changed.length} changed fields, only composite and renderer leaf`);
}
