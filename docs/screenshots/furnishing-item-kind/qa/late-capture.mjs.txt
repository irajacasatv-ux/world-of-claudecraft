async function press(page, variant, selector) {
  await page.waitForSelector(selector, { visible: true, timeout: 15000 });
  await page.$eval(selector, (element) => element.scrollIntoView({ block: 'nearest', inline: 'nearest' }));
  if (variant.mobile) await page.tap(selector);
  else await page.click(selector);
  await settle();
}
async function visibleTarget(page, selector) {
  await page.$eval(selector, (element) => element.scrollIntoView({
    block: 'nearest', inline: 'nearest', behavior: 'instant',
  }));
  await settle();
  return page.$eval(selector, (element) => {
    const box = element.getBoundingClientRect();
    const x = box.left + box.width / 2;
    const y = box.top + box.height / 2;
    if (box.width <= 0 || box.height <= 0 || box.left < 0 || box.top < 0
      || box.right > innerWidth + 1 || box.bottom > innerHeight + 1
      || !element.contains(document.elementFromPoint(x, y))) throw new Error(`Changed target not fully visible: ${element.outerHTML.slice(0, 200)}`);
    return { bounds: box.toJSON(), text: element.textContent.trim() };
  });
}
async function captureNoticesAbsent(page) {
  return page.evaluate(() => {
    return ['#gpu-notice', '#perf-nudge'].every((selector) => {
      const notice = document.querySelector(selector);
      if (!notice) return true;
      const style = getComputedStyle(notice);
      const box = notice.getBoundingClientRect();
      return style.display === 'none' || style.visibility === 'hidden' || box.width === 0 || box.height === 0;
    });
  });
}
async function dismissLateCaptureNotices(page) {
  // Match the screenshot target pipeline's live dismiss button path. The
  // entry helper seeds future documents; a late preview boot can rearm this
  // notice in the current document. The performance nudge carries similar
  // software-rendering copy and can arrive later, so dismiss both here.
  const dismissedPerfNudge = await dismissPerfNudge(page);
  const clicked = await page.evaluate(() => {
    const clicked = [];
    for (const selector of ['.gpu-notice-dismiss']) {
      const dismiss = document.querySelector(selector);
      if (dismiss && !dismiss.closest('#gpu-notice')?.hidden) {
        dismiss.click(); clicked.push(selector);
      }
    }
    return clicked;
  });
  if (dismissedPerfNudge) clicked.push('.perf-nudge-dismiss');
  await settle(80);
  assert.equal(await captureNoticesAbsent(page), true, 'GPU or performance notice still covers the capture');
  return { dismissedThroughButton: clicked, visibleBeforeScreenshot: false };
}
async function shot(page, variant, name, selector, result) {
  const files = [`${variant.key}-${name}.png`, `${variant.key}-${name}-detail.png`];
  const noticeAttempts = [];
  // The perf nudge's first live-frame check can land inside screenshot capture.
  // Its real dismiss button ends that session's notice. Keep rejected pixels
  // and repeat only this known notice race, never a failed content assertion.
  for (let attempt = 1; attempt <= 3; attempt++) {
    const captureNotices = await dismissLateCaptureNotices(page);
    const full = await page.screenshot();
    if (!(await captureNoticesAbsent(page))) {
      const rejected = `${variant.key}-${name}-rejected-${attempt}-full.png`;
      fs.writeFileSync(`${output}/${rejected}`, full);
      noticeAttempts.push({ attempt, ...captureNotices,
        rejected: [rejected], reason: 'notice appeared during full screenshot' });
      continue;
    }
    const bounds = await page.$eval(selector, (element) => {
      const box = element.getBoundingClientRect();
      const scrolling = [...element.querySelectorAll('*')].filter((node) => node.scrollTop || node.scrollLeft).map((node) => ({ tag: node.tagName, className: node.className, scrollTop: node.scrollTop, scrollLeft: node.scrollLeft }));
      return { scrolling, x: Math.max(0, box.left), y: Math.max(0, box.top),
        right: Math.min(innerWidth, box.right), bottom: Math.min(innerHeight, box.bottom),
        unclipped: box.left >= 0 && box.top >= 0 && box.right <= innerWidth + 1 && box.bottom <= innerHeight + 1 };
    });
    assert.ok(bounds.right > bounds.x && bounds.bottom > bounds.y, `Visible capture region missing: ${name}`);
    if (expectFixed && selector === '#tooltip' && name !== 'paperdoll-control-card') assert.equal(bounds.unclipped, true, `Fixed tooltip clipped: ${name}`);
    const detail = await page.screenshot({ clip: { x: bounds.x, y: bounds.y,
      width: bounds.right - bounds.x, height: bounds.bottom - bounds.y } });
    if (!(await captureNoticesAbsent(page))) {
      const rejected = [`${variant.key}-${name}-rejected-${attempt}-full.png`,
        `${variant.key}-${name}-rejected-${attempt}-detail.png`];
      fs.writeFileSync(`${output}/${rejected[0]}`, full);
      fs.writeFileSync(`${output}/${rejected[1]}`, detail);
      noticeAttempts.push({ attempt, ...captureNotices,
        rejected, reason: 'notice appeared during detail screenshot' });
      continue;
    }
    fs.writeFileSync(`${output}/${files[0]}`, full);
    fs.writeFileSync(`${output}/${files[1]}`, detail);
    if (page.__qaTouchHeld) {
      await page.touchscreen.touchEnd();
      page.__qaTouchHeld = false;
    }
    return { name, ...result, bounds, captureNotices: { ...captureNotices,
      acceptedAttempt: attempt, noticeAttempts }, screenshots: files };
  }
  report.diagnostics.push({ variant: variant.key, name, noticeAttempts });
  assert.fail(`GPU or performance notice interrupted all capture attempts: ${name}`);
}
async function hoverCard(page, variant, selector, expected, forbidden = []) {
  await page.evaluate(() => window.__game.hud.hideTooltip());
  await page.$eval(selector, (element) => element.scrollIntoView({ block: 'nearest', inline: 'nearest' }));
  await settle();
  const point = await page.evaluate((selector) => {
    const element = document.querySelector(selector);
    const box = element.getBoundingClientRect();
    const x = box.left + Math.min(22, box.width / 2);
    const y = box.top + box.height / 2;
    if (!element.contains(document.elementFromPoint(x, y))) throw new Error(`Hover target occluded: ${selector}`);
    window.__qaLateNativeEvents = [];
    for (const type of ['pointerdown', 'mouseenter']) document.addEventListener(type, (event) => {
      if (event.composedPath().some((node) => node instanceof Element && node.matches(selector))) {
        window.__qaLateNativeEvents.push({ type: event.type, trusted: event.isTrusted, pointerType: event.pointerType ?? null });
      }
    }, { capture: true });
    return { x, y };
  }, selector);
  if (variant.mobile) {
    await page.touchscreen.touchStart(point.x, point.y);
    page.__qaTouchHeld = true;
  }
  else { await page.mouse.move(0, 0); await page.mouse.move(point.x, point.y); }
  try {
    await page.waitForFunction(({ expected, mobile }) => {
      const tooltip = document.querySelector('#tooltip');
      return tooltip && getComputedStyle(tooltip).display !== 'none'
        && expected.every((word) => tooltip.textContent.includes(word))
        && window.__qaLateNativeEvents.some((event) => event.trusted && (mobile
          ? event.type === 'pointerdown' && event.pointerType === 'touch' : event.type === 'mouseenter'));
    }, { timeout: 10000 }, { expected, mobile: variant.mobile });
    const card = await page.evaluate((forbidden) => {
      const element = document.querySelector('#tooltip');
      const text = element.textContent;
      for (const word of forbidden) if (text.includes(word)) throw new Error(`Forbidden tooltip claim: ${word}`);
      return { lines: [...element.children].map((node) => node.textContent.trim()), text,
        events: window.__qaLateNativeEvents, html: element.innerHTML };
    }, forbidden);
    return card;
  } catch (error) {
    if (page.__qaTouchHeld) {
      await page.touchscreen.touchEnd();
      page.__qaTouchHeld = false;
    }
    throw error;
  }
}

async function capture(page, variant) {
  const injected = await fixture(page);
  await settle(3000);
  await page.waitForFunction(() => {
    const loading = document.querySelector('#loading-screen');
    return !loading || getComputedStyle(loading).display === 'none' || getComputedStyle(loading).visibility === 'hidden';
  }, { timeout: 180000 });
  await dismissEntryOverlays(page);
  const captures = [];
  await page.evaluate(() => window.__game.hud.toggleWocMarket());
  await page.waitForSelector('#woc-market-window [data-tt-key="browse:1"]', { visible: true, timeout: 15000 });
  const rarity = async (selector) => page.$eval(selector, (element, fixed) => {
    const icon = element.matches('img') ? element : element.querySelector('img');
    if (!icon?.complete || !icon.naturalWidth) throw new Error('Item icon not decoded');
    const expected = fixed ? 'q-rare' : 'q-legendary';
    if (!icon.classList.contains(expected)) throw new Error(`Wrong furnishing rarity ${icon.className}, expected ${expected}`);
    return { text: element.textContent.trim(), iconClass: icon.className, accessibleName: element.getAttribute('aria-label') };
  }, expectFixed);
  const browse = await rarity('#woc-market-window img[data-tt-key="browse:1"]');
  browse.name = await page.$eval('#woc-market-window .wm-name[data-tt-key="browse:1"]', (element) => element.textContent.trim());
  browse.accessibleName = await page.$eval('#woc-market-window [data-focus-key="wm-row-1"]', (element) => element.getAttribute('aria-label'));
  assert.equal(browse.name, 'Screenshot Test Furnishing');
  assert.ok(browse.accessibleName.includes('Screenshot Test Furnishing'), 'Exchange row accessible name is false');
  const controlBrowse = await page.$eval('#woc-market-window img[data-tt-key="browse:2"]', (icon) => icon.classList.contains('q-legendary'));
  assert.equal(controlBrowse, true, 'Eligible Exchange control loses its rolled tier');
  captures.push(await shot(page, variant, 'exchange-browse', '#woc-market-window', { furnishing: browse, controlLegendary: true, target: await visibleTarget(page, '#woc-market-window [data-focus-key="wm-row-1"]') }));
  await press(page, variant, '#woc-market-window [data-tab="sell"]');
  await press(page, variant, '#woc-market-window [data-field="sell-search"]');
  await page.waitForSelector('#woc-market-window [data-sell-index="0"]', { visible: true, timeout: 10000 });
  const sell = await rarity('#woc-market-window [data-sell-index="0"]');
  const controlSell = await page.$eval('#woc-market-window [data-sell-index="1"] img', (icon) => icon.classList.contains('q-legendary'));
  assert.equal(controlSell, true, 'Eligible Sell control loses its rolled tier');
  captures.push(await shot(page, variant, 'exchange-sell', '#woc-market-window', { furnishing: sell, controlLegendary: true, target: await visibleTarget(page, '#woc-market-window [data-sell-index="0"]') }));
  const exchangeCard = await hoverCard(page, variant, '#woc-market-window [data-sell-index="0"] img',
    ['Screenshot Test Furnishing', 'Rare Furnishing', 'Footprint: 2 by 3 cells.'], ['Forged Furnishing Name', 'Legendary Furnishing']);
  captures.push(await shot(page, variant, 'exchange-furnishing-card', '#tooltip', exchangeCard));

  await page.evaluate(() => { window.__game.hud.wocMarketWindow.close(); window.__game.hud.openMarket(); const bags = document.querySelector('#bags'); if (bags) bags.style.display = 'none'; });
  await press(page, variant, '#market-window [data-tab="collect"]');
  const ledger = await page.evaluate((fixed) => {
    const names = [...document.querySelectorAll('#market-window .mkt-sale-name > span:first-child')].map((node) => node.textContent.trim());
    const expected = [fixed ? 'Screenshot Test Furnishing' : 'Forged Furnishing Name', 'Control Blade'];
    if (JSON.stringify(names) !== JSON.stringify(expected)) throw new Error(`Wrong Collect names: ${JSON.stringify(names)}`);
    const buyers = [...document.querySelectorAll('#market-window .mkt-sale-buyer')].map((node) => node.textContent.trim());
    if (buyers.length !== 2 || buyers.some((name) => !name.includes('Buyer'))) throw new Error('Collect buyer control missing');
    return { names, buyers, displayedText: document.querySelector('#market-window').textContent };
  }, expectFixed);
  ledger.target = await visibleTarget(page, '#market-window .mkt-sale');
  captures.push(await shot(page, variant, 'market-collect', '#market-window', ledger));

  await page.evaluate(() => { window.__game.hud.closeMarket(); window.__game.hud.toggleChar(); const bags = document.querySelector('#bags'); if (bags) bags.style.display = 'none'; window.__game.hud.syncCharBagsPairing(); });
  await page.waitForSelector('#char-window .char-mw-slots-value', { visible: true, timeout: 15000 });
  const paperdoll = await page.evaluate((fixed) => {
    const value = document.querySelector('#char-window .char-mw-slots-value').textContent.trim();
    if (value !== (fixed ? '1 / 2' : '2 / 2')) throw new Error(`Wrong Masterwrought cap: ${value}`);
    const chest = document.querySelector('#equip-slot-chest');
    const gear = document.querySelector('#equip-slot-mainhand');
    if (Boolean(chest.querySelector('.equip-mw-chip')) !== !fixed) throw new Error('Furnishing chip mismatch');
    if (!gear.querySelector('.equip-mw-chip')) throw new Error('Eligible Masterwrought control chip missing');
    if (chest.querySelector('.slot-item').textContent.trim() !== 'Screenshot Test Furnishing') throw new Error('Furnishing authored worn name changed');
    if (!chest.querySelector('img.q-rare')) throw new Error('Furnishing authored worn rarity missing');
    return { value, furnishingChip: !fixed, controlChip: true, chestName: chest.querySelector('.slot-item').textContent.trim() };
  }, expectFixed);
  await page.$eval('#char-window .char-mw-slots-value', (element) => element.scrollIntoView({ block: 'nearest' }));
  captures.push(await shot(page, variant, 'paperdoll-cap', '#char-window', { ...paperdoll, target: await visibleTarget(page, '#char-window .char-mw-slots-value') }));
  await page.$eval('#equip-slot-chest', (element) => element.scrollIntoView({ block: 'nearest' }));
  captures.push(await shot(page, variant, 'paperdoll-chest', '#char-window', { ...paperdoll, target: await visibleTarget(page, '#equip-slot-chest') }));
  const capText = await page.evaluate((fixed) => fixed ? window.__qaLateUi.capUsed1 : window.__qaLateUi.capUsed2, expectFixed);
  const chestCard = await hoverCard(page, variant, '#equip-slot-chest',
    ['Screenshot Test Furnishing', 'Made by Testmaker.', ...(expectFixed ? [] : [capText])],
    expectFixed ? ['Masterwrought', 'Strength', 'Legendary', 'Forged Furnishing Name'] : []);
  captures.push(await shot(page, variant, 'paperdoll-furnishing-card', '#tooltip', chestCard));
  const controlCard = await hoverCard(page, variant, '#equip-slot-mainhand', [capText]);
  captures.push(await shot(page, variant, 'paperdoll-control-card', '#tooltip', controlCard));
  const readCalls = await page.evaluate(() => window.__qaLateUi.readCalls);
  assert.ok(readCalls.some((call) => call.method === 'status') && readCalls.some((call) => call.method === 'browse'), 'Actual Exchange read flow was not exercised');
  return { variant, injected, captures, readCalls };
}

try {
  for (const variant of variants) {
    console.log(`PROGRESS ${variant.key}: opening ${expectFixed ? 'fixed' : 'baseline'} archive`);
    const page = await browser.newPage();
    page.on('pageerror', (error) => { report.failures.push(`${variant.key} pageerror: ${error.message}`); });
    page.on('response', (response) => { if (response.status() >= 400) report.diagnostics.push({ variant: variant.key, status: response.status(), path: new URL(response.url()).pathname }); });
    page.on('console', (message) => { if (message.type() === 'error') report.diagnostics.push({ variant: variant.key, message: message.text() }); });
    try {
      if (variant.mobile) await page.emulate({
        viewport: { width: variant.width, height: variant.height, isMobile: true, hasTouch: true, deviceScaleFactor: 2 },
        userAgent: 'Mozilla/5.0 (iPhone; CPU iPhone OS 17_0 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/17.0 Mobile/15E148 Safari/604.1',
      });
      await suppressGpuNotice(page);
      await page.evaluateOnNewDocument(() => {
        localStorage.removeItem('woc_entry_probe');
        const settings = JSON.parse(localStorage.getItem('woc_settings') || '{}');
        settings.graphicsPreset = 1; settings.graphicsDefaultApplied = true;
        localStorage.setItem('woc_settings', JSON.stringify(settings));
      });
      await page.goto(gameUrl, { waitUntil: 'networkidle0', timeout: navTimeoutMs });
      if (variant.mobile) await page.evaluate(() => document.body.classList.add('mobile-touch'));
      const entered = await enterOfflineGame(page, { charName: 'Testmaker', charClass: 'warrior', settleMs: 3000,
        selectorTimeoutMs, gameBootTimeoutMs: 180000 });
      assert.equal(entered, true, 'Offline entry failed');
      const result = await capture(page, variant);
      report.variants.push(result);
      console.log(`PASS ${variant.key}: ${result.captures.length} actual-window captures`);
    } catch (error) {
      report.failures.push(`${variant.key}: ${error.stack ?? error.message}`);
      await page.screenshot({ path: `${output}/${variant.key}-failure.png` }).catch(() => {});
      console.error(`FAIL ${variant.key}: ${error.message}`);
    } finally { await page.close(); }
  }
} finally {
  await browser.close();
  fs.writeFileSync(`${output}/manifest.json`, `${JSON.stringify(report, null, 2)}\n`);
}
console.log(`Captured ${report.variants.length}/${variants.length}; failures ${report.failures.length}. Manifest: ${output}/manifest.json`);
process.exit(report.variants.length === variants.length && report.failures.length === 0 ? 0 : 1);
